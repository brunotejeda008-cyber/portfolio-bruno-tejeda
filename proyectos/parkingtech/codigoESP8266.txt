// --- Configuración de Pines ---

// Plaza 1

const int trigPin1 = D2;

const int echoPin1 = D1;

const int ledPin1 = D0;



// Plaza 2

const int trigPin2 = D3;

const int echoPin2 = D4;

const int ledPin2 = D7;



void setup() {

Serial.begin(9600);


// Configurar pinos como Entradas o Salidas

pinMode(trigPin1, SALIDA);

pinMode(echoPin1, INPUT);

pinMode(ledPin1, SALIDA);



pinMode(trigPin2, SALIDA);

pinMode(echoPin2, INPUT);

pinMode(ledPin2, SALIDA);

}

void loop() {
